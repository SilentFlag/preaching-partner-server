# Database Schema
# TODO: Should the user_group_pair contain perms instead of the service_group table?

CREATE TABLE congregation(id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL, updated INTEGER, deleted BOOL);

CREATE TABLE categories(id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL, prefix VARCHAR(255) NOT NULL, congregation INTEGER, updated INTEGER, CONSTRAINT fk_congregation FOREIGN KEY (congregation) REFERENCES congregation(id));

CREATE TABLE service_group(id INTEGER PRIMARY KEY, name VARCHAR(255) NOT NULL, congregation INTEGER, elder INTEGER, updated INTEGER, deleted BOOL, CONSTRAINT fk_congregation FOREIGN KEY (congregation) REFERENCES congregation(id), CONSTRAINT fk_elder FOREIGN KEY (elder) REFERENCES users(id));

CREATE TABLE users(id INTEGER PRIMARY KEY, firstname VARCHAR(255) NOT NULL, lastname VARCHAR(255) NOT NULL, password VARCHAR(255), enabled BOOL NOT NULL, activation_code INT, updated INTEGER);

CREATE TABLE user_cong_pair(user_id INTEGER, congregation_id INTEGER, deleted BOOL, updated INTEGER, CONSTRAINT fk_congregation FOREIGN KEY (congregation_id) REFERENCES congregation(id), CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES users(id));

CREATE TABLE user_group_pair(user_id INTEGER, group_id INTEGER, deleted BOOL, updated INTEGER, CONSTRAINT fk_group_id FOREIGN KEY (group_id) REFERENCES service_group(id), CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES users(id));

CREATE TABLE maps(id INTEGER PRIMARY KEY, assignee INT, assigner INT, category INT NOT NULL, visibility INT NOT NULL, file_name VARCHAR(255), updated INTEGER, congregation_id INTEGER, CONSTRAINT fk_category FOREIGN KEY (category) REFERENCES categories(id), CONSTRAINT fk_congregation FOREIGN KEY (congregation_id) REFERENCES congregation(id));

CREATE TABLE tokens(user INTEGER NOT NULL, refresh BOOL NOT NULL, token VARCHAR(255) NOT NULL, CONSTRAINT fk_user FOREIGN KEY (user) REFERENCES users(id));

CREATE TABLE street(id INTEGER PRIMARY KEY, map_id, name VARCHAR(255), CONSTRAINT fk_map_id FOREIGN KEY (map_id) REFERENCES maps(id));

CREATE TABLE addresses(id INTEGER PRIMARY KEY, street_id INTEGER, number VARCHAR(255), tags BLOB, visited BOOL, CONSTRAINT fk_street_id FOREIGN KEY (street_id) REFERENCES street(id));
